#!/bin/bash
apt update && apt install -y frr
sed -i 's/ospfd=no/ospfd=yes/' /etc/frr/daemons
sed -i 's/eigrpd=no/eigrpd=yes/' /etc/frr/daemons
cat <<EOF > /etc/frr/frr.conf
interface enp0s3
 ip address 200.0.0.18/30
interface enp0s8
 ip address 200.0.0.21/30
router ospf
 ospf router-id 1.1.1.4
 network 200.0.0.16/30 area 2
 network 200.0.0.20/30 area 2
 redistribute eigrp 10
router eigrp 10
 network 200.0.0.20/30
 redistribute ospf
EOF
systemctl restart frr

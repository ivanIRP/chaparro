#!/bin/bash
# CONFIGURACIÓN CLIENTE JC (9.0.0.20)
# Ejecutar como root

cat <<EOF > /etc/network/interfaces
auto enp0s3
iface enp0s3 inet static
    address 9.0.0.20
    netmask 255.255.255.0
EOF
systemctl restart networking

# Añadir el servidor al archivo hosts
echo "9.0.0.100 mail.acme.net" >> /etc/hosts

apt update && apt install -y thunderbird

echo "--- MÁQUINA DE JC LISTA ---"
echo "Usa jc@acme.net / pass: 123"

#!/bin/bash
# Ejecutar como root en el Cliente 2

# 1. Configuración de IP
cat <<EOF > /etc/network/interfaces
auto enp0s3
iface enp0s3 inet static
    address 9.0.0.20
    netmask 255.255.255.0
EOF
systemctl restart networking

# 2. Instalación de Thunderbird
apt update && apt install -y thunderbird

echo "Configuración completada para jc@acme.net"

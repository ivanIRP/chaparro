#!/bin/bash
apt update && apt install -y frr
sed -i 's/ospfd=no/ospfd=yes/' /etc/frr/daemons
cat <<EOF > /etc/frr/frr.conf
interface enp0s3
 ip address 200.0.0.10/30
interface enp0s8
 ip address 200.0.0.13/30
router ospf
 ospf router-id 1.1.1.2
 network 200.0.0.8/30 area 0
 network 200.0.0.12/30 area 1
EOF
systemctl restart frr

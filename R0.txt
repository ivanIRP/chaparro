#!/bin/bash
# Instalación
apt update && apt install -y frr traceroute
echo "net.ipv4.ip_forward=1" >> /etc/sysctl.conf && sysctl -p

# Configuración de interfaces
# enp0s3 -> Puente (Teléfono) - Usará DHCP para integrarse a tu red real
# enp0s8 -> Interna (Hacia R1)
# enp0s9 -> Interna (Hacia C1)

sed -i 's/ripd=no/ripd=yes/' /etc/frr/daemons

cat <<EOF > /etc/frr/frr.conf
interface enp0s3
 description SALIDA_TELEFONO
 ip address dhcp
!
interface enp0s8
 ip address 200.0.0.1/30
!
interface enp0s9
 ip address 10.0.0.1/29
!
router rip
 network enp0s8
 network enp0s9
 redistribute connected
EOF

systemctl restart frr

apt update && apt install -y frr
sed -i 's/ospfd=no/ospfd=yes/' /etc/frr/daemons
sed -i 's/ripd=no/ripd=yes/' /etc/frr/daemons
sed -i 's/eigrpd=no/eigrpd=yes/' /etc/frr/daemons
echo "net.ipv4.ip_forward=1" >> /etc/sysctl.conf
sysctl -p
systemctl restart frr

cat <<EOF > /etc/frr/frr.conf
interface enp0s3
 ip address 200.0.0.6/30
interface enp0s8
 ip address 200.0.0.2/30
interface enp0s9
 ip address 10.0.0.9/29
router ospf
 ospf router-id 1.1.1.0
 network 200.0.0.4/30 area 0
 redistribute rip metric 20
router rip
 version 2
 network 200.0.0.0/30
 network 10.0.0.8/29
 redistribute ospf metric 1
 redistribute connected
EOF
systemctl restart frr

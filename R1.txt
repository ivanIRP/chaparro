#!/bin/bash
apt update && apt install -y frr
echo "net.ipv4.ip_forward=1" >> /etc/sysctl.conf && sysctl -p
sed -i 's/ripd=no/ripd=yes/' /etc/frr/daemons
sed -i 's/ospfd=no/ospfd=yes/' /etc/frr/daemons
cat <<EOF > /etc/frr/frr.conf
interface enp0s3
 ip address 200.0.0.2/30
interface enp0s8
 ip address 200.0.0.5/30
interface enp0s9
 ip address 10.0.0.9/29
router rip
 network 200.0.0.0/30
 network 10.0.0.8/29
 redistribute ospf
router ospf
 network 200.0.0.4/30 area 0
 redistribute rip
EOF
systemctl restart frr

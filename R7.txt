#!/bin/bash
apt update && apt install -y frr traceroute
echo "net.ipv4.ip_forward=1" >> /etc/sysctl.conf && sysctl -p

sed -i 's/eigrpd=no/eigrpd=yes/' /etc/frr/daemons

cat <<EOF > /etc/frr/frr.conf
interface enp0s3
 description SALIDA_TELEFONO
 ip address dhcp
!
interface enp0s8
 ip address 200.0.0.26/30
!
interface enp0s9
 ip address 10.0.0.25/29
!
router eigrp 10
 network 200.0.0.24/30
 network 10.0.0.24/29
 redistribute connected
EOF

systemctl restart frr

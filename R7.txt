apt update && apt install -y frr
sed -i 's/ospfd=no/ospfd=yes/' /etc/frr/daemons
sed -i 's/ripd=no/ripd=yes/' /etc/frr/daemons
sed -i 's/eigrpd=no/eigrpd=yes/' /etc/frr/daemons
echo "net.ipv4.ip_forward=1" >> /etc/sysctl.conf
sysctl -p
systemctl restart frr

cat <<EOF > /etc/frr/frr.conf
interface enp0s3
 ip address 200.0.0.26/30
interface enp0s9
 ip address 10.0.0.25/29
router eigrp 10
 network 200.0.0.24/30
 network 10.0.0.24/29
 redistribute connected
EOF
systemctl restart frr

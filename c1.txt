#!/bin/bash
# CONFIGURACIÓN CLIENTE IVAN (9.0.0.10)
# Ejecutar como root

cat <<EOF > /etc/network/interfaces
auto enp0s3
iface enp0s3 inet static
    address 9.0.0.10
    netmask 255.255.255.0
EOF
systemctl restart networking

# Añadir el servidor al archivo hosts para que resuelva el nombre
echo "9.0.0.100 mail.acme.net" >> /etc/hosts

apt update && apt install -y evolution

echo "--- MÁQUINA DE IVAN LISTA ---"
echo "Usa ivan@acme.net / pass: 123"
